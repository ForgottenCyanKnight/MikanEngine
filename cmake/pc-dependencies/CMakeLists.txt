cmake_minimum_required(VERSION 3.30)
project(MikanPCDependencies LANGUAGES C CXX)
if(NOT MSVC OR NOT WIN32)
    message(FATAL_ERROR "PC dependency package currently supports Windows/MSVC only")
endif()
set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
add_compile_options(/utf-8 /FS /Zi)
add_compile_definitions(NOMINMAX)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" upper)
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/lib")
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
if(NOT MIKAN_SOURCE_ROOT)
    get_filename_component(MIKAN_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(dep "${MIKAN_SOURCE_ROOT}/dependencies")
file(GLOB_RECURSE jolt CONFIGURE_DEPENDS "${dep}/JoltPhysics/Jolt/*.cpp")
add_library(JoltPhysics STATIC ${jolt})
target_include_directories(JoltPhysics PUBLIC "${dep}/JoltPhysics")
target_compile_definitions(JoltPhysics PRIVATE JPH_DEBUG_RENDERER=0 JPH_PROFILE_ENABLED=0 JPH_DISABLE_TEMP_ALLOCATOR_NAMES=1 JPH_USE_CUSTOM_ALLOCATOR=0)
file(GLOB msdf CONFIGURE_DEPENDS "${dep}/msdfgen/core/*.cpp")
add_library(msdfgen STATIC ${msdf})
target_include_directories(msdfgen PUBLIC "${dep}")
file(GLOB_RECURSE box CONFIGURE_DEPENDS "${dep}/box2d/src/*.c")
add_library(Box2D STATIC ${box})
target_compile_options(Box2D PRIVATE /std:c11 /D_Atomic=)
target_include_directories(Box2D PRIVATE "${dep}/box2d/include" "${dep}/box2d/src" "${dep}/box2d/msvc_compat" "${dep}/box2d/extern/simde")
add_library(ImGui STATIC
    "${dep}/imgui/imgui.cpp" "${dep}/imgui/imgui_demo.cpp"
    "${dep}/imgui/imgui_draw.cpp" "${dep}/imgui/imgui_tables.cpp"
    "${dep}/imgui/imgui_widgets.cpp" "${dep}/imgui/imgui_impl_sdl3.cpp"
    "${dep}/imgui/imgui_impl_vulkan.cpp")
target_include_directories(ImGui PRIVATE "${dep}" "${dep}/imgui" "${dep}/SDL3" "${dep}/vulkan")
# Export directives survive static linking into the sole owner, Game.dll.
target_compile_definitions(ImGui PRIVATE "IMGUI_API=__declspec(dllexport)" "IMGUI_IMPL_API=__declspec(dllexport)")
if(MIKAN_ENABLE_NRD)
    set(NRD_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
    set(NRD_NRI OFF CACHE BOOL "" FORCE)
    set(NRD_EMBEDS_SPIRV_SHADERS ON CACHE BOOL "" FORCE)
    set(NRD_EMBEDS_DXIL_SHADERS OFF CACHE BOOL "" FORCE)
    set(NRD_EMBEDS_DXBC_SHADERS OFF CACHE BOOL "" FORCE)
    set(NRD_SUPPORTS_QUAD_INTRINSICS OFF CACHE BOOL "" FORCE)
    set(NRD_NORMAL_ENCODING 4 CACHE STRING "" FORCE)
    set(NRD_ROUGHNESS_ENCODING 1 CACHE STRING "" FORCE)
    set(NRD_SHADERS_PATH "${CMAKE_BINARY_DIR}/nrd-shaders" CACHE STRING "" FORCE)
    add_subdirectory("${dep}/NRD" nrd)
    set_target_properties(NRD PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(NRD PROPERTIES ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
    endforeach()
endif()
